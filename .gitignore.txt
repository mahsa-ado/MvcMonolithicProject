-- ایجاد دیتابیس
CREATE DATABASE DataBaseClass
GO

USE DataBaseClass;
GO

--  ناشران
CREATE TABLE publishers (
    pub_id      NVARCHAR(4)     NOT NULL PRIMARY KEY,
    pub_name    NVARCHAR(40)    NULL,
    city        NVARCHAR(20)    NULL,
    state       NVARCHAR(30)     NULL,
    country     NVARCHAR(30)    NULL 
);

--  اطلاعات ناشر
CREATE TABLE pub_info (
    pub_id      NVARCHAR(4)     NOT NULL PRIMARY KEY,
    logo        VARBINARY(MAX)  NULL,
    pr_info     NVARCHAR(MAX)   NULL,
    CONSTRAINT FK_pub_info_publishers FOREIGN KEY (pub_id) REFERENCES publishers(pub_id)
);

--  مشاغل
CREATE TABLE jobs (
    job_id      SMALLINT PRIMARY KEY       IDENTITY(1,1) ,
    job_desc    NVARCHAR(50)    NOT NULL ,
    min_lvl     TINYINT         NOT NULL,
    max_lvl     TINYINT         NOT NULL
);

--  کارمندان
CREATE TABLE employee (
    emp_id      NVARCHAR(9)     NOT NULL PRIMARY KEY,
    fname       NVARCHAR(20)    NOT NULL,
    minit       NVARCHAR(1)     NULL,
    lname       NVARCHAR(30)    NOT NULL,
    job_id      SMALLINT        NOT NULL ,
    job_lvl     TINYINT         NULL  ,
    pub_id      NVARCHAR(4)     NOT NULL ,
    hire_date   DATETIME        NOT NULL DEFAULT GETDATE(),
    CONSTRAINT FK_employee_jobs FOREIGN KEY (job_id) REFERENCES jobs(job_id),
    CONSTRAINT FK_employee_publishers FOREIGN KEY (pub_id) REFERENCES publishers(pub_id)
);

-- جدول نویسندگان
CREATE TABLE authors (
    au_id       NVARCHAR(11)    NOT NULL PRIMARY KEY,
    au_lname    NVARCHAR(40)    NOT NULL,
    au_fname    NVARCHAR(20)    NOT NULL,
    phone       NVARCHAR(12)    NOT NULL ,
    address     NVARCHAR(40)    NULL,
    city        NVARCHAR(20)    NULL,
    state       NVARCHAR(2)     NULL,
    zip         NVARCHAR(5)     NULL,
    contract    BIT             NOT NULL
);

-- جدول کتاب‌ها
CREATE TABLE titles (
    title_id    NVARCHAR(6)     NOT NULL PRIMARY KEY,
    title       NVARCHAR(80)    NOT NULL,
    type        NVARCHAR(12)    NOT NULL DEFAULT N'UNDECIDED',
    pub_id      NVARCHAR(4)     NULL,
    price       MONEY           NULL,
    advance     MONEY           NULL,
    royalty     INT             NULL,
    ytd_sales   INT             NULL,
    notes       NVARCHAR(200)   NULL,
    pubdate     DATETIME        NOT NULL DEFAULT GETDATE(),
    CONSTRAINT FK_titles_publishers FOREIGN KEY (pub_id) REFERENCES publishers(pub_id)
);

-- جدول رابطه نویسنده و عنوان
CREATE TABLE titleauthor (
    au_id       NVARCHAR(11)    NOT NULL,
    title_id    NVARCHAR(6)     NOT NULL,
    au_ord      TINYINT         NULL,
    royaltyper  INT             NULL,
    PRIMARY KEY (au_id, title_id),
    CONSTRAINT FK_titleauthor_authors FOREIGN KEY (au_id) REFERENCES authors(au_id),
    CONSTRAINT FK_titleauthor_titles FOREIGN KEY (title_id) REFERENCES titles(title_id)
);

-- جدول فروشگاه‌ها
CREATE TABLE stores (
    stor_id     NVARCHAR(4)     NOT NULL PRIMARY KEY,
    stor_name   NVARCHAR(40)    NULL,
    stor_address NVARCHAR(40)   NULL,
    city        NVARCHAR(20)    NULL,
    state       NVARCHAR(2)     NULL,
    zip         NVARCHAR(5)     NULL
);

-- جدول فروش
CREATE TABLE sales (
    stor_id     NVARCHAR(4)     NOT NULL,
    ord_num     NVARCHAR(20)    NOT NULL,
    ord_date    DATETIME        NOT NULL,
    qty         SMALLINT        NOT NULL,
    payterms    NVARCHAR(12)    NOT NULL,
    title_id    NVARCHAR(6)     NOT NULL,
    PRIMARY KEY (stor_id, ord_num, title_id),
    CONSTRAINT FK_sales_stores FOREIGN KEY (stor_id) REFERENCES stores(stor_id),
    CONSTRAINT FK_sales_titles FOREIGN KEY (title_id) REFERENCES titles(title_id)
);

-- جدول تخفیف‌ها
CREATE TABLE discounts (
    discounttype NVARCHAR(40)   NOT NULL,
    stor_id     NVARCHAR(4)     NULL,
    lowqty      SMALLINT        NULL,
    highqty     SMALLINT        NULL,
    discount    MONEY    NOT NULL,
    CONSTRAINT FK_discounts_stores FOREIGN KEY (stor_id) REFERENCES stores(stor_id)
);

-- جدول  حق‌التألیف
CREATE TABLE roysched (
    title_id    NVARCHAR(6)     NOT NULL,
    lorange     INT             NULL,
    hirange     INT             NULL,
    royalty     INT             NULL,
    CONSTRAINT FK_roysched_titles FOREIGN KEY (title_id) REFERENCES titles(title_id)
);
GO
